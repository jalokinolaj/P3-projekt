Heuristic Function"manhattan"(node_a, node_b, coordinates)
    (x_1, y_1)=coordinates[node_a]
    (x_2, y_2)=coordinates[node_b]
    h = absolute(x_1-x_2)+ absolute(y_1-y_2)
    return h

function reconstructPath(came_from, current_state)
    total_path = [current_state]
    while current_state in came_from
        current_state = came_from[current_state]
        total_path.prepend(current_state)
    return total_path
